#include <Arduino.h>
#include <Wire.h>               // ไลบรารีสำหรับการสื่อสารแบบ I2C (ใช้เชื่อมต่อกับจอ OLED)
#include <Adafruit_GFX.h>       // ไลบรารีจัดการกราฟิกพื้นฐาน (วาดเส้น พิมพ์ตัวหนังสือ)
#include <Adafruit_SSD1306.h>   // ไลบรารีควบคุมหน้าจอ OLED รุ่น SSD1306 โดยเฉพาะ
#include <Preferences.h>        // ไลบรารีสำหรับบันทึกค่าลงหน่วยความจำถาวร (เหมือนแฟลชไดร์ฟเล็กๆ ปิดเครื่องค่าไม่หาย เอาไว้ใช้กับปุ่ม SAVE)
#include <SPIFFS.h>             // ไลบรารีจัดการระบบไฟล์ (สร้างฮาร์ดดิสก์จำลองในบอร์ด เพื่อเก็บไฟล์ log)
#include <FS.h>                 // ไลบรารีจัดการโครงสร้างไฟล์ (ทำงานร่วมกับ SPIFFS เอาไว้อ่าน/เขียนไฟล์)
#include <ArduinoJson.h>        // ไลบรารีสำหรับสร้างและอ่านข้อมูลรูปแบบ JSON (แพ็กเกจข้อมูลที่เอาไว้คุยกับหน้าเว็บ)

// 🔵 นำเข้าไลบรารีของ Bluetooth
// ---------------------------------------------------------
// ทำหน้าที่แปลง ESP32 ให้ปล่อยสัญญาณบลูทูธคลาสสิก (Bluetooth Classic)
#include "BluetoothSerial.h"

// โค้ดส่วนนี้คือการเช็คระบบบอร์ด ป้องกันคนลืมเปิดฟังก์ชัน Bluetooth ตอนลงโปรแกรม
// ถ้าบอร์ดไม่ได้รองรับหรือไม่ได้เปิดระบบบลูทูธไว้ โค้ดจะหยุดทำงานและฟ้อง Error สีแดงๆ ทันที
#if !defined(CONFIG_BT_ENABLED) || !defined(CONFIG_BLUEDROID_ENABLED)
#error Bluetooth is not enabled! Please run `make menuconfig` to and enable it
#endif

// ---------------- 1. ⚙️ ตั้งค่าขาอุปกรณ์ฮาร์ดแวร์ ----------------
#define POT_PIN 36   // ขา Analog รับค่าจากตัวต้านทานปรับค่าได้ (แกนหมุน)
#define LED_PIN 5    // ขา Digital สำหรับควบคุมไฟ LED
#define SW2_PIN 32   // ขาปุ่มกดที่ 1 (ในที่นี้ใช้กดเพื่อเปิด/ปิด LED)
#define SW3_PIN 33   // ขาปุ่มกดที่ 2 (ใช้กดเพื่อดึงค่าหมุน POT ไปตั้งเป็นความสว่าง PWM)
#define SW4_PIN 34   // ขาปุ่มกดที่ 3 (กดสั้น=เปิด/ปิด Log, กดค้าง=ลบไฟล์ Log ทิ้ง)

// ---------------- 2. 🔵 ตัวแปรสำหรับ Bluetooth ----------------
// สร้างออบเจกต์ชื่อ "SerialBT" เอาไว้สำหรับรับ-ส่งข้อมูลล่องหนผ่านบลูทูธ
BluetoothSerial SerialBT; 
String student_id = "6730300655_6730300809"; // รหัสนิสิต เอาไว้ต่อท้ายชื่อบลูทูธป้องกันการแย่งกันต่อในห้องเรียน

// ---------------- 3. 🖥️ ตัวแปรหน้าจอ OLED ----------------
#define SCREEN_WIDTH 128 // กำหนดความกว้างจอ (พิกเซล)
#define SCREEN_HEIGHT 64 // กำหนดความสูงจอ (พิกเซล)
// สร้างออบเจกต์จอ OLED เชื่อมต่อผ่าน I2C (&Wire)
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);

// ---------------- 4. 🧠 ตัวแปรระบบและสถานะต่างๆ ----------------
Preferences preferences; // สร้างออบเจกต์สำหรับบันทึกและอ่านค่าถาวร (ใช้งานตอนกด SAVE)
bool ledState = false;   // เก็บสถานะหลอดไฟปัจจุบัน (false = ปิด, true = เปิด)
uint8_t pwmValue = 0;    // เก็บค่าความสว่างของหลอดไฟ (0-255)
bool loggingState = false; // สถานะระบบบันทึกไฟล์ (false = ไม่จดลงไฟล์, true = กำลังจดลงไฟล์)
unsigned long logCount = 0; // ตัวนับว่าตอนนี้จดข้อมูลลงไฟล์ SPIFFS ไปแล้วกี่บรรทัด

// ตัวแปรสำหรับ "จำค่าที่ถูกกด SAVE" แยกออกมา เพื่อส่งไปโชว์ที่หน้าเว็บในกล่อง PREFERENCES ให้ถูกต้อง
bool savedLedState = false;
uint8_t savedPwmValue = 0;

// ---------------- 5. 💡 ตั้งค่าสัญญาณ PWM (ความสว่างหลอดไฟ) ----------------
const int pwmChannel = 0;      // ช่องสัญญาณ PWM (ESP32 มีหลายช่อง เลือกใช้ช่อง 0)
const int pwmFreq = 5000;      // ความถี่ของสัญญาณ (5000 Hz)
const int pwmResolution = 8;   // ความละเอียด 8-bit (คือปรับความสว่างได้ 256 ระดับ 0-255)

// ---------------- 6. ⏱️ ตัวแปรสำหรับเช็คการกดปุ่ม (Debounce) ----------------
bool lastSW2 = HIGH, lastSW3 = HIGH, lastSW4 = HIGH; // เก็บสถานะปุ่มกดก่อนหน้า (ค่าปกติปุ่มที่ไม่ได้กดคือ HIGH)
unsigned long pressTimeSW4 = 0;  // จับเวลาตอนเริ่มกดปุ่ม SW4 เพื่อคำนวณว่ากดค้างไหม
bool isPressingSW4 = false;      // เช็คสถานะว่า SW4 กำลังถูกแช่อยู่หรือไม่
unsigned long lastDebounceTime = 0;     // จับเวลาป้องกันปุ่มเบิ้ล (กด 1 ครั้งแต่นับ 2 ครั้ง)
const unsigned long debounceDelay = 50; // เวลาที่ใช้หน่วงปุ่มกด (50 มิลลิวินาที)
unsigned long lastBtTxTime = 0;  // จับเวลาสำหรับการส่งข้อมูลบลูทูธ Real-time ทุก 1 วินาที


// ==============================================================================
// 💾 ฟังก์ชัน: ระบบบันทึกไฟล์ (ฮาร์ดดิสก์ SPIFFS)
// ตำแหน่งที่เก็บ: อยู่ในแฟลชเมมโมรี่ของบอร์ด (SPIFFS)
// ชื่อไฟล์: "/log.csv"
// รูปแบบไฟล์: ไฟล์ข้อความธรรมดา คั่นด้วยลูกน้ำ (CSV) เอาไปเปิดใน Excel ได้เลย
// ==============================================================================
void appendLog(int adc, int pwm, float vol, bool state) {
  if (!loggingState) return; // ถ้าสถานะ Log=OFF ให้หยุดทำงานฟังก์ชันนี้ทันที ไม่จดลงไฟล์
  
  // เปิดไฟล์ชื่อ "/log.csv" โหมด FILE_APPEND (คือเขียนข้อมูลต่อท้ายบรรทัดล่างสุดไปเรื่อยๆ)
  File file = SPIFFS.open("/log.csv", FILE_APPEND);
  if (file) {
    unsigned long t = millis() / 1000; // คำนวณเวลาที่บอร์ดทำงานหน่วยเป็นวินาที
    char timeStr[6];
    sprintf(timeStr, "%02ld:%02ld", (t / 60) % 100, t % 60); // แปลงวินาที เป็นฟอร์แมต MM:SS
    
    char logLine[64];
    // นำตัวแปรต่างๆ มาต่อกันด้วยลูกน้ำ สร้างเป็น 1 บรรทัด เช่น "01:25,2048,128,1.65,ON"
    sprintf(logLine, "%s,%d,%d,%.2f,%s", timeStr, adc, pwm, vol, state ? "ON" : "OFF");
    
    file.println(logLine); // สั่งจดบรรทัดนั้นลงไฟล์
    file.close();          // ปิดไฟล์ (ต้องปิดเสมอเพื่อบันทึกลงดิสก์จริงๆ)
    logCount++;            // บวกตัวนับจำนวนบรรทัดไปอีก 1
  }
}

// ==============================================================================
// 🤖 ฟังก์ชัน: จัดการคำสั่ง (รับคำสั่งบลูทูธแล้วแปลผล)
// ==============================================================================
void processCommandAction(String cmd, int val = -1) {
  int potValue = analogRead(POT_PIN);          // แอบอ่านค่าหมุน POT ปัจจุบันไว้ก่อน
  float vol = (potValue / 4095.0) * 3.3;       // แปลงค่า POT เป็นแรงดันไฟ 0-3.3V

  // 1. ถ้าคำสั่งคือควบคุม LED
  if (cmd == "led") {
    if (val != -1) ledState = (val == 1); // รับคำสั่งจากบลูทูธ (val คือค่าที่เว็บส่งมา)
    else ledState = !ledState;            // รับคำสั่งจากปุ่มกดบนบอร์ด (สลับเปิด/ปิด)
    
    // สั่งจ่ายไฟความสว่าง PWM (ใช้ 255-pwmValue เพราะหลอดไฟต่อแบบ Active-Low)
    if (ledState) ledcWrite(pwmChannel, 255 - pwmValue); else ledcWrite(pwmChannel, 255);
    appendLog(potValue, pwmValue, vol, ledState); // จดลงไฟล์ด้วยถ้าระบบ LOG เปิดอยู่
  } 
  // 2. ถ้าคำสั่งคือ เปิด/ปิด การจดไฟล์ (LOG)
  else if (cmd == "logging") {
    if (val != -1) loggingState = (val == 1);
    else loggingState = !loggingState;
  } 
  // 3. ถ้าคำสั่งคือ ตั้งค่าความสว่าง PWM
  else if (cmd == "pwm" && val >= 0) {
    pwmValue = val; // เอาค่าที่เว็บส่งมาไปจำไว้
    if (ledState) ledcWrite(pwmChannel, 255 - pwmValue); // สั่งไฟให้สว่างตามค่าใหม่ทันที
    appendLog(potValue, pwmValue, vol, ledState);
  } 
  // 4. ถ้าคำสั่งคือ SAVE
  else if (cmd == "save") {
    // นำค่า LED และ PWM ปัจจุบันเขียนลง Preferences (หน่วยความจำถาวร)
    preferences.putBool("ledState", ledState);
    preferences.putUChar("pwmValue", pwmValue);
    // อัปเดตตัวแปรจำค่า เพื่อเอาไปส่งให้เว็บวาดลงกล่อง PREFERENCES
    savedLedState = ledState;
    savedPwmValue = pwmValue;
    Serial.println("Saved to Preferences!");
  } 
  // 5. ถ้าคำสั่งคือ ลบไฟล์ Log
  else if (cmd == "deleteLog") {
    SPIFFS.remove("/log.csv"); // ลบไฟล์เก่าทิ้ง
    File file = SPIFFS.open("/log.csv", FILE_WRITE); // สร้างไฟล์ใหม่เอี่ยม
    if (file) { file.println("Time,ADC,PWM,Voltage,LED_State"); file.close(); } // เขียนหัวตารางทิ้งไว้
    logCount = 0; // รีเซ็ตตัวนับบรรทัดเป็น 0
  } 
  // ---------------------------------------------------------
  // 🔵 6. ถ้าคำสั่งคือ READLOG (ยิงไฟล์ Log กลับไปหาคอมพิวเตอร์)
  // ---------------------------------------------------------
  else if (cmd == "readlog") {
    File file = SPIFFS.open("/log.csv", "r"); // เปิดไฟล์แบบ "r" (Read/อ่าน)
    if (!file) return;
    
    // ยิงคำว่า "LOG_BEGIN" ทะลุบลูทูธไปบอกหน้าเว็บว่า "เตรียมรับไฟล์ยาวๆ นะ!"
    SerialBT.println("LOG_BEGIN"); 
    
    // วนลูปอ่านข้อมูลในไฟล์ทีละบรรทัดจนกว่าจะหมด
    while (file.available()) {
      String line = file.readStringUntil('\n'); // อ่าน 1 บรรทัด
      line.trim();
      if(line.length() > 0) {
        // ยิงข้อมูลบรรทัดนั้นลอยผ่านบลูทูธไปเข้าหน้าเว็บ
        SerialBT.println(line); 
      }
    }
    // ยิงคำว่า "LOG_END" เพื่อบอกหน้าเว็บว่า "ส่งครบหมดแล้ว เอาไปโชว์ได้เลย"
    SerialBT.println("LOG_END"); 
    file.close(); // ปิดไฟล์
  }
}

// ==============================================================================
// 🚀 ฟังก์ชัน: SETUP (ทำงานครั้งเดียวตอนเริ่มเปิดบอร์ด)
// ==============================================================================
void setup() {
  Serial.begin(115200); // เปิดการสื่อสารผ่านสาย USB เอาไว้ให้ช่างคอมดู Log 

  // ---------------------------------------------------------
  // 🔵 การตั้งชื่อและเปิดสวิตช์ปล่อยสัญญาณบลูทูธ
  // ---------------------------------------------------------
  // นำคำว่า "LAB10A-" มาต่อด้วยรหัสนิสิต เพื่อตั้งเป็น "ชื่ออุปกรณ์ (Device Name)"
  String btName = "LAB10A-" + student_id; 
  
  // สั่งบอร์ดเปิดบลูทูธคลาสสิกด้วยชื่อที่เราตั้งไว้ (คอมพิวเตอร์จะสแกนเจอชื่อนี้)
  SerialBT.begin(btName); 
  Serial.println("Bluetooth Started! Ready to pair...");

  // ตั้งค่าขาปุ่มกด (INPUT) และหลอดไฟ (OUTPUT)
  pinMode(SW2_PIN, INPUT); pinMode(SW3_PIN, INPUT); pinMode(SW4_PIN, INPUT);
  pinMode(LED_PIN, OUTPUT);
  // ผูกช่องสัญญาณ PWM เข้ากับขาไฟ LED
  ledcSetup(pwmChannel, pwmFreq, pwmResolution);
  ledcAttachPin(LED_PIN, pwmChannel);

  // สตาร์ทหน้าจอ OLED ให้พร้อมทำงาน
  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) Serial.println(F("OLED failed"));
  display.clearDisplay(); display.setTextColor(WHITE); // ตั้งฟอนต์สีขาว

  // โหลดค่า Preferences (ค่าที่เคย SAVE ถาวรไว้ก่อนโดนถอดปลั๊ก) กลับมา
  preferences.begin("ledPref", false); 
  ledState = preferences.getBool("ledState", false); 
  pwmValue = preferences.getUChar("pwmValue", 0);    
  savedLedState = ledState;
  savedPwmValue = pwmValue;
  // สั่งให้หลอดไฟสว่างตามค่าที่โหลดมาทันที
  if (ledState) ledcWrite(pwmChannel, 255 - pwmValue); else ledcWrite(pwmChannel, 255);

  // 💾 สตาร์ทระบบไฟล์ SPIFFS (เตรียมฮาร์ดดิสก์บนบอร์ด)
  if (!SPIFFS.begin(true)) {
    Serial.println("SPIFFS Mount Failed"); 
  } else {
    if (!SPIFFS.exists("/log.csv")) { // เช็คว่ามีไฟล์ชื่อ /log.csv อยู่ในดิสก์ไหม
      File file = SPIFFS.open("/log.csv", FILE_WRITE); // ถ้าไม่มี ให้สร้างใหม่
      file.println("Time,ADC,PWM,Voltage,LED_State"); file.close();
    } else {
      File file = SPIFFS.open("/log.csv", FILE_READ); // ถ้ามีอยู่แล้ว ให้เปิดมานับบรรทัด
      if (file) {
        file.readStringUntil('\n'); // ข้ามหัวตารางบรรทัดแรกไป
        while (file.available()) {
          String line = file.readStringUntil('\n');
          line.trim();
          if (line.length() > 0) logCount++; // เจอข้อมูล 1 บรรทัดก็บวก logCount ไป 1
        }
        file.close();
      }
    }
  }
}

// ==============================================================================
// 🔄 ฟังก์ชัน: LOOP (วนรอบทำงานตลอดเวลา)
// ==============================================================================
void loop() {
  int potValue = analogRead(POT_PIN);               // ดึงค่าหมุน POT ปัจจุบัน
  float currentVoltage = (potValue / 4095.0) * 3.3; // แปลงเป็นโวลต์
  unsigned long currentMillis = millis();           // ดึงเวลาการทำงานเพื่อเอาไว้ใช้ตั้งจังหวะ (Timer)

  // ---------------------------------------------------------
  // 🔵 1. ส่วนรับคำสั่งจากบลูทูธ (คอมพิวเตอร์ -> ESP32)
  // ---------------------------------------------------------
  // SerialBT.available() ทำหน้าที่เป็นพนักงานคอยเช็คว่า "มีคอมพิวเตอร์ส่งคำสั่งอะไรมาหาเราไหม?"
  if (SerialBT.available()) {
    // ถ้ามี ให้อ่านข้อความนั้นจนกว่าจะจบ 1 บรรทัด (เจอตัว Enter หรือ \n) เอามาเก็บในตัวแปร btCmd
    String btCmd = SerialBT.readStringUntil('\n'); 
    btCmd.trim();         // ลบเว้นวรรคส่วนเกินทิ้ง ป้องกันบั๊ก
    btCmd.toUpperCase();  // แปลงทุกตัวอักษรเป็นตัวพิมพ์ใหญ่ (เช่น led 1 -> LED 1) จะได้เช็คง่ายๆ

    // ถอดรหัสคำสั่งที่อ่านมาได้
    if (btCmd.startsWith("LED ")) {            // หน้าเว็บสั่งให้เปิด/ปิดหลอดไฟ
      processCommandAction("led", btCmd.substring(4).toInt()); 
    } else if (btCmd.startsWith("PWM ")) {     // หน้าเว็บเลื่อนสไลเดอร์ปรับความสว่าง
      processCommandAction("pwm", btCmd.substring(4).toInt());
    } else if (btCmd.startsWith("LOG ")) {     // หน้าเว็บกดเปิด/ปิดระบบจดไฟล์
      processCommandAction("logging", btCmd.substring(4).toInt());
    } else if (btCmd == "SAVE") {              // หน้าเว็บกดปุ่ม SAVE
      processCommandAction("save");
    } else if (btCmd == "READLOG") {           // หน้าเว็บกดปุ่มขอดูไฟล์ Log (เรียกบรรทัดที่ 113)
      processCommandAction("readlog");
    } else if (btCmd == "DELLOG") {            // หน้าเว็บกดปุ่มลบไฟล์ (ถังขยะ)
      processCommandAction("deleteLog");
    }
  }

  // ---------------------------------------------------------
  // 🔵 2. ส่วนส่งข้อมูล Real-Time (ESP32 -> หน้าเว็บ HTML)
  // ---------------------------------------------------------
  // เงื่อนไข: ส่งข้อมูลทุกๆ 1000 มิลลิวินาที (1 วินาที) เป๊ะๆ
  if (currentMillis - lastBtTxTime >= 1000) {
    lastBtTxTime = currentMillis; // รีเซ็ตตัวจับเวลาใหม่
    
    // เตรียมกล่องข้อมูลแบบ JSON (ฟอร์แมตข้อมูลมาตรฐานที่หน้าเว็บใช้)
    StaticJsonDocument<256> doc;
    doc["lab"] = "10A";                   // บอกเว็บว่าเป็นแลป 10A
    doc["student"] = student_id;
    doc["adc"] = potValue;                // ส่งค่า ADC (เว็บเอาเลขนี้ไปวาดกราฟเส้นสีน้ำเงิน 🔵)
    doc["voltage"] = currentVoltage;
    doc["led"] = ledState;                // ส่งสถานะ LED ไปโชว์อักษร ON/OFF
    doc["pwm"] = pwmValue;                // ส่งค่า PWM (เว็บเอาเลขนี้ไปวาดกราฟเส้นสีเขียว 🟢)
    doc["savedLED"] = savedLedState;      // ส่งค่าไฟที่เพิ่งเซฟถาวรไปโชว์ในกล่อง PREFERENCES
    doc["savedPWM"] = savedPwmValue;      // ส่งค่า PWM ที่เพิ่งเซฟถาวรไปโชว์ในกล่อง PREFERENCES
    doc["logging"] = loggingState;
    doc["logCount"] = logCount;           // ส่งว่าตอนนี้ไฟล์ log มีกี่บรรทัดแล้ว
    doc["uptime"] = currentMillis / 1000; // เวลาการทำงานหน่วยวินาที

    String jsonResponse;
    serializeJson(doc, jsonResponse);     // สั่งแพ็กกล่องข้อมูลทั้งหมดนี้ ให้กลายเป็นตัวอักษรเรียงกัน 1 บรรทัด
    
    Serial.println(jsonResponse);         // ปริ้นท์ออกสาย USB เผื่อคนเขียนโปรแกรมจะเปิดดูเอง
    
    // 🔴 จุดยิงข้อมูล! 🔴
    // เช็คก่อนว่า "ตอนนี้มีคอมพิวเตอร์ Pair และเชื่อมต่อหน้าเว็บค้างไว้อยู่ไหม?"
    if (SerialBT.hasClient()) {
      // ถ้ายืนยันว่ามีคนดูอยู่ ก็สั่งยิงก้อน JSON บรรทัดเดียวลอยไปตามคลื่นบลูทูธเลย!
      // พอคอมพิวเตอร์ได้รับ หน้าเว็บ HTML จะรับข้อมูลชุดนี้ไปแกะกล่อง แล้วเอาตัวเลขไปเติมในกราฟให้เราดูทันที
      SerialBT.println(jsonResponse);
    }
  }

  // ---------------- 3. ⚙️ ส่วนควบคุมด้วยปุ่มกดฮาร์ดแวร์ (สวิตช์บนบอร์ด) ----------------
  bool sw2 = digitalRead(SW2_PIN);
  bool sw3 = digitalRead(SW3_PIN);
  bool sw4 = digitalRead(SW4_PIN);

  // ป้องกันปุ่มเบิ้ล (ถ้ากดปุ่มห่างกันเกิน 50 มิลลิวินาทีถึงจะยอมรับว่าตั้งใจกด)
  if (currentMillis - lastDebounceTime > debounceDelay) {
    if (sw2 == LOW && lastSW2 == HIGH) { // กดปล่อยปุ่ม SW2 = สลับไฟ LED
      processCommandAction("led");
      lastDebounceTime = currentMillis;
    }
    if (sw3 == LOW && lastSW3 == HIGH) { // กดปล่อยปุ่ม SW3 = แปลงค่าหมุน POT ไปเป็น PWM ทันที
      processCommandAction("pwm", map(potValue, 0, 4095, 0, 255));
      lastDebounceTime = currentMillis;
    }
    
    // เช็คการกดค้างปุ่ม SW4
    if (sw4 == LOW && lastSW4 == HIGH) { pressTimeSW4 = currentMillis; isPressingSW4 = true; } // จังหวะนิ้วกดลงไป
    
    if (isPressingSW4 && sw4 == LOW) { // นิ้วกำลังแช่ปุ่มอยู่
      if (currentMillis - pressTimeSW4 >= 3000) { // ถ้านับเวลาแช่ปุ่มเกิน 3000ms (3 วินาที)
        processCommandAction("deleteLog");        // สั่งลบไฟล์ดิสก์ล้างบางเลย
        isPressingSW4 = false;                    // ลบเสร็จสับสวิตช์กันมันลบซ้ำ
      }
    }
    if (sw4 == HIGH && lastSW4 == LOW) { // จังหวะปล่อยนิ้วออกจากปุ่ม
      if (isPressingSW4) { 
        unsigned long pressDuration = currentMillis - pressTimeSW4;
        if (pressDuration > 50 && pressDuration < 3000) processCommandAction("logging"); // ถ้ากดสั้นๆ ให้สลับเปิด/ปิดจดไฟล์ Log 
      }
      isPressingSW4 = false;
      lastDebounceTime = currentMillis;
    }
  }
  lastSW2 = sw2; lastSW3 = sw3; lastSW4 = sw4; // จำสถานะปุ่มเอาไว้เทียบในลูปถัดไป

  // ---------------- 4. 🖥️ ส่วนอัปเดตหน้าจอ OLED ----------------
  static unsigned long lastDisplayTime = 0;
  // วาดหน้าจอใหม่ทุกๆ 100 มิลลิวินาที (ถ้าวาดบ่อยกว่านี้จอจะกระตุก)
  if (currentMillis - lastDisplayTime > 100) {
    display.clearDisplay();  // เอายางลบลบจอให้ดำสนิท
    display.setCursor(0, 0); // ไปเริ่มเขียนที่มุมซ้ายบนของจอ
    
    int currentPwmNow = map(potValue, 0, 4095, 0, 255); // คำนวณค่า PWM จำลองตามแกน POT ณ ตอนนั้น
    
    // ปริ้นท์บรรทัด 1: แสดง Uptime (นาที:วินาที) และค่าการหมุน POT
    display.printf("%02ld:%02ld    POT: %04d\n\n", (currentMillis / 1000 / 60) % 100, (currentMillis / 1000) % 60, potValue);
    
    // ปริ้นท์บรรทัด 2 และ 3: แสดงค่า PWM แบบสดๆ และค่า PWM ที่แอบ SAVE ถาวรไว้
    display.printf("NOW : %03d (%.2fV)\n", currentPwmNow, (currentPwmNow / 255.0) * 3.3);
    display.printf("SAVE: %03d (%.2fV)\n\n", savedPwmValue, (savedPwmValue / 255.0) * 3.3);
    
    // ---------------------------------------------------------
    // 🔵 อัปเดตสถานะการเชื่อมต่อบลูทูธโชว์ลงจอ OLED
    // ---------------------------------------------------------
    // เอาพนักงาน (SerialBT.hasClient) มาเช็คอีกรอบ
    if (SerialBT.hasClient()) display.printf("BT: Connected\n"); // ถ้ามีคอมพิวเตอร์กำลังสูบข้อมูลอยู่ ให้โชว์ Connected
    else display.printf("BT: Waiting...\n"); // ถ้ายังไม่มีคนเปิดเว็บมาดู ปริ้นท์ว่า Waiting... (รออยู่จ้า)
    
    // ปริ้นท์บรรทัด 4: แสดงสถานะไฟ LED ตอนนี้ และจำนวนบรรทัดที่จดอยู่ในไฟล์ Log
    display.printf("LED:%s | LOG:%04ld\n", ledState ? "ON " : "OFF", logCount);
    
    display.display(); // สั่งให้ข้อความทั้งหมดที่เราปริ้นท์ไว้ ปรากฏขึ้นมาบนจอ OLED พร้อมกัน
    lastDisplayTime = currentMillis; // รีเซ็ตตัวจับเวลาการวาดจอ
  }
}
