src/lie/ml_model.cpp
float predict(const Model& m, const Result& r) {
float x[FEAT_COUNT];
extract(r, x);            
// 12 feature
double z = m.b;
for (int i = 0; i < m.n; i++)
z += m.w[i] * (x[m.idx[i]] - m.mean[i])
 / m.scale[i];
return 1.0 / (1.0 + exp(-z));   // sigmoid
}
